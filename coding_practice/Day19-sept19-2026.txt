Ansible - Setup Promethus for Monitoring
-name: Install Prometheus for Monitoring
 hosts: all
 become: yes
 task:
   -name: Install Promethus dependencies
    yum:
      name:
      	- prometheus
	- prometheus-node-exporter
	state: present
   -name: Start Prometheus service
    service:
	name:prometheus
	state:started
	enabled:yes

Terraform - Setup AWS Cloudwatch for Monitoring "aws_cloudwatch_log_group" "my_log_group" { name = "my-log-group"}


resource "aws_cloudwatch_log_stream" "my_log_stream" {                          log_group_name = aws_cloudwatch_log_group.my_log_group.name
name             = "my-log-stream"
}

Kubernetes - Deploy Fluentd for Logging

apiVersion: app/v1
kind: Deployment
metadata:
 name: fluentd
spec:
 replicas:1
 selector:
  matchLabels:
	app:fluentd
  spec:
	containers:
	-name: fluentd
	image: fluent/fluentd
	ports:
	- containerPort: 24224

---
apiVersion:v1
Kind:service
metadata:
	name: fluentd
     spec:
	selector:
	app:fluentd
     ports:
     - protocol: TCP
       port:24224

 Jenkinsfile - Intergrating Prometheus Monitoring in Pipeline
Pipeline{
	agent any
	stages { stage ('Build') {
steps {
echo 'Building Docker image...'
docker.build('my-web-app')}
 }
}
stage ('Test') {
steps{
echo 'Running test...'} 
}
stage('Monitor Deployment') {
steps {
	script {
		sh 'kubectl apply-f kubernetes/deployment/yaml'
		sh 'kubectl get pods --watch'
	}
}
