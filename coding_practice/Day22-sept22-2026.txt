Day 22 — Multi-Cluster Kubernetes Management

Anisble - deploy Nginx Across Multiple kubernetes Clusters

#switch to a cluster , deploy. then repeat for the second cluster

 - name: Deploy Nginx across multiple kubernetes clusters
 hosts: localhost
 tasks:
  - name: Configure kubectl for cluster1
    command: kubectl config use-context us-east-cluster

  - name: Deploy Ngnix on cluster 1
    command: kubectl apply -f kubernetes/deployment.yaml

  - name: Configure kubectl for cluster2
    command: kubectl config use-context us-west-cluster

  - name: Deploy Ngnix on cluster2 
    command: kubectl apply -f kubernetes/deployment.yaml

Kubernetes deployment for one cluster

 #normal deployment for one cluster

 apiVersion: apps/v1
 kind: Deployment
 metadata:
  name: himanshi-deployment
 spec:
 replicas: 3
 selector:
  matchLabels:
    app: Himanshi-app
  template:
      metadata:
        labels:
          app: Himanshi-app
   spec:
      containers:
        - name : nginx
          image: nginx  
          ports: 
            - containerPort: 80

---

#federated deployment: same app, for many clusters

 apiVersion: federation.k8s.io/v1beta1
 kind: FederatedDeployment
 metadata:

    name: Himanshi-federated-deployment
    namespace: default 

  spec:
    template:
      spec:
        replicas: 3
       selector:
        matchlabels:
          app: himanshi-app

        spec:
          containers:
          -name: nginx
            image: nginx
            ports:
              - containerPort:80



Jenkinsfile - MultiCluster Deployment Strategy 

// one stage per cluster
 
 pipeline {
    
    agent any 

    stages {
          stage('Deploy to Cluster 1')  {
              steps {  

                  script {            
                        sh 'kubectl --context=us-east-cluster apply -f kubernetes/deployment.yaml' 

                }                       
             }

           }  
        }
  }
