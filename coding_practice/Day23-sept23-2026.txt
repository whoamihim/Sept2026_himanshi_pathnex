 Ansible - Create a Lambda function 

 - name: My first Lambda
   hosts: localhost
   connection: local
   tasks:
      - name: Check AWS CLI is installed
        command: aws --version
 
      - name: Zip the code
        command: zip app.zip app.py
        
      - name: create the lambda function 
        command: aws lambda create-function --function-name himanshi-hello --runtime 



 Terraform - Lambda + Functio URL (simple serverless API)

#1. IAM role: give lambda permission to run 

 resource: "aws_iam_role" "lambda_role" { 
    name: "himanshi-lambda-role" "lambda_role"
    assume_role_policy: jasonencode ({    
        Version = "2012-10-17"
        Statement = [{ 
          Effect  = "Allow"
          Principal = { Service = "lambda.amazonawsaws.com"}
          Action    = "sts.AssumeRole"
            }] 
         })           
      }



#2. The Lambda function 
  

  resource "aws_lambda_function" "hello"  {
    function_name = "himanshi-hello"
    runtime       = "pyhton3.12"
    handler       = "app.handler"
    filename      = "app.zip" 
    role          = "aws_iam_role.lambda_role.arn"   
}


#3. Functiom URL : a public Link for the lambda

 resource "aws_lambda_function_url" "hello_url" {
     function_name    = aws_lambda_function.hello.function_name
     authorization_type = "NONE"  
    }


# Print the link after terraform apply

 output "my_api_link"  {
     value = aws_lambda_function_url.hello_url.function_url       }



 Kubernetes - Deploy a microservice with Helm

 helm create my-service             # create a ready-made chart
 helm install my-app ./my-service   # deploy it to the cluster
 helm list                          # see what is running
 helm uninstall my-app              # remove it


 Jenkinsfile - Lambda update pipeline

 pipeline {
      agent any 
      stages {
          stage('Zip') {
              steps {   
                      sh 'zip app.zip app.py'   
                     } 
          } 
            stage('Deploy') {
                steps{
          
                    sh 'aws lambda update-function-code --function-name himanshi-hello'
             }
    
           }

        }
    }

 Docker - Multi-Stage Build 

 #stage1: Build the code (big image, user only for building)

 FROM maven:3.9-eclipse-temurin-17 AS build
 WORKDIR /app
 COPY . .
 RUN mvn package


 #stage2: take only the .war file (small image, for production)

 FROM tomcat:9
 COPY --from=build /app/target/*.war /usr/local/tomcat/webapps/ EXPOSE 8080
